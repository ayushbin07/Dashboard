import React from "https://esm.sh/react@19";
import { createRoot } from "https://esm.sh/react-dom@19/client";
import confetti from "https://esm.sh/canvas-confetti@1";

function App() {
  const triggerConfetti = (event, color) => {
    const button = event.currentTarget;
    const rect = button.getBoundingClientRect();
    const x = (rect.left + rect.width / 2) / window.innerWidth;
    const y = (rect.top + rect.height / 2) / window.innerHeight;

    confetti({
      particleCount: 100,
      spread: 360,
      startVelocity: 30,
      origin: { x, y },
      ...(color ? { colors: [color] } : {})
    });
  };

  return (
    <div className="flex items-center justify-center min-h-screen bg-slate-900">
      <div className="space-x-4">
        <button
          onClick={(e) => triggerConfetti(e, "#800080")}
          className="px-6 py-3 bg-slate-900 border border-slate-500 text-lg text-white rounded-full hover:bg-slate-800 hover:scale-105 active:scale-95 transition-all"
        >
          🟣 Single Color
        </button>
        <button
          onClick={(e) => triggerConfetti(e)}
          className="px-6 py-3 bg-slate-900 border border-slate-500 text-lg text-white rounded-full hover:bg-slate-800 hover:scale-105 active:scale-95 transition-all"
        >
          🎉 Party
        </button>
      </div>
    </div>
  );
}

createRoot(document.getElementById("root")).render(<App />);
